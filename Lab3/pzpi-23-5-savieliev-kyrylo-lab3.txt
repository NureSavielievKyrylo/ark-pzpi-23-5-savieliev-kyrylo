private createDumpStream(): {
stream: NodeJS.ReadableStream
waitForDump: Promise<void>
} {
const { user, password, host, port, database } = this.config.db
const pgDump = spawn(
'pg_dump',

'-U',
user,
'-d',
database,
'--no-password',
'--format=plain',
],

const waitForDump = new Promise<void>((resolve, reject) => {
let stderrOutput = ''
pgDump.stderr.on('data', (data: Buffer) => {
stderrOutput += data.toString()
})
pgDump.on('close', (code) => {
if (code === 0) {
resolve()
} else {
reject(new PgDumpError(stderrOutput.trim() || `Exit code:
${code}`))
}
})
pgDump.on('error', (err) => {
reject(new PgDumpError(err.message))
})
})
return {
stream: pgDump.stdout,
waitForDump,
}
}

Також було розроблено бізнес логіку для аналізу історії зміни статусів
рукописів за допомогою матриці переходів Маркова, виявлення аномалій активності
користувачів, баєсівського передбачення дедлайнів (див. рис. Б.2) та симуляції
Монте-Карло для оцінки часу завершення роботи над рукописами.
Приклад функції для генерації випадкових чисел за логнормальним
розподілом:
export const randomLogNormal = (mean: number, stdDev: number): number
=> {
const variance = stdDev * stdDev
const mu = Math.log((mean * mean) / Math.sqrt(variance + mean *
mean))
const sigma = Math.sqrt(Math.log(1 + variance / (mean * mean)))
return Math.exp(randomNormal(mu, sigma))
}

4 ВИСНОВКИ
Під час виконання лабораторної роботи було розроблено бізнес логіку
та функції адміністрування серверної частини програмної системи. Реалізовані
функції адміністрування включають логування активності, резервне копіювання та
відновлення даних у БД, блокування/розблокування користувачів та переривання
їх сесій.
Розроблена бізнес логіка включає аналіз історії зміни статусів рукописів
за допомогою матриці переходів Маркова, виявлення аномалій активності
користувачів, баєсівське передбачення дедлайнів та симуляцію Монте-Карло для
оцінки часу завершення роботи над рукописами. Всі розроблені функції були
протестовані та успішно інтегровані у серверну частину програмної системи.
